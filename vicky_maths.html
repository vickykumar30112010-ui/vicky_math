// Apni Gemini API key yahan dalein const API_KEY = "AAPKI_GEMINI_API_KEY_YAHAN_DALEIN"; async function solveMathPhoto(imageFile) { // Image ko Base64 format mein convert karein const base64Image = await convertToBase64(imageFile); const promptText = "Is math question ko solve karein aur step-by-step solution Hindi/English mein dein."; const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=${API_KEY}`, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ contents: [{ parts: [ { text: promptText }, { inline_data: { mime_type: imageFile.type, data: base64Image } } ] }] }) }); const data = await response.json(); const solution = data.candidates[0].content.parts[0].text; // Solution ko screen par display karein document.getElementById("solution-box").innerText = solution; } // Helper function to convert file to base64 function convertToBase64(file) { return new Promise((resolve, reject) => { const reader = new FileReader(); reader.readAsDataURL(file); reader.onload = () => resolve(reader.result.split(',')[1]); reader.onerror = error => reject(error); }); }
